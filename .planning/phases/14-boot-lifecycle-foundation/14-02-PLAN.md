# Plan 14-02: Shared HTTP Client with Connection Pooling

<action>
Update `ingestclient.New()` to configure a pooled `http.Transport`:

```go
package ingestclient

import (
    "bytes"
    "context"
    "crypto/tls"
    "fmt"
    "net/http"
    "time"
)

// Client handles telemetry delivery to the central ingestion service.
type Client struct {
    url    string
    client *http.Client
}

// New creates a new mTLS Ingestion Client with a production-grade pooled transport.
// The client is safe for concurrent use and reuses connections across heartbeats and events.
func New(url string, tlsConfig *tls.Config) *Client {
    transport := &http.Transport{
        TLSClientConfig:     tlsConfig,
        MaxIdleConns:        10,
        MaxIdleConnsPerHost: 5,
        IdleConnTimeout:     90 * time.Second,
        DisableCompression:  false,
    }
    return &Client{
        url: url,
        client: &http.Client{
            Transport: transport,
            Timeout:   10 * time.Second,
        },
    }
}

// Post sends a signed telemetry envelope to the ingestion service.
// On failure, it returns an error; the caller determines the drop/retry policy.
func (c *Client) Post(ctx context.Context, payload []byte) error {
    req, err := http.NewRequestWithContext(ctx, "POST", c.url, bytes.NewBuffer(payload))
    if err != nil {
        return fmt.Errorf("failed to create request: %w", err)
    }
    req.Header.Set("Content-Type", "application/json")

    resp, err := c.client.Do(req)
    if err != nil {
        return fmt.Errorf("ingestion service unreachable: %w", err)
    }
    defer resp.Body.Close()

    if resp.StatusCode != http.StatusAccepted && resp.StatusCode != http.StatusOK {
        return fmt.Errorf("ingestion service rejected request: %s", resp.Status)
    }

    return nil
}

// HTTPClient exposes the underlying *http.Client for OOB or health-check callers
// that share this transport rather than creating their own http.Client.
func (c *Client) HTTPClient() *http.Client {
    return c.client
}
```

```go
// sendOOBAlert fires a JSON alert to the OOB webhook using the shared ingest client transport.
func sendOOBAlert(ctx context.Context, ic *ingestclient.Client, url string, payload map[string]interface{}) {
    payload["ts"] = time.Now().UTC().Format(time.RFC3339Nano)
    body, err := json.Marshal(payload)
    if err != nil {
        log.Printf("[oob] Failed to marshal alert payload: %v", err)
        return
    }

    req, err := http.NewRequestWithContext(ctx, http.MethodPost, url, bytes.NewReader(body))
    if err != nil {
        log.Printf("[oob] Failed to create OOB request: %v", err)
        return
    }
    req.Header.Set("Content-Type", "application/json")

    resp, err := ic.HTTPClient().Do(req)
    if err != nil {
        log.Printf("[oob] OOB alert delivery failed: %v", err)
        return
    }
    resp.Body.Close()
}
```

Update all call sites in `main.go` to pass `ctx` and `ic`:
- `sendOOBAlert(ctx, ic, cfg.OOBAlert.URL, map[string]interface{}{...})`

The old signature `sendOOBAlert(url string, payload map[string]interface{})` must be removed.
