# Plan 27-01: Multi-Tenant Platform and Dashboard API

## Objective

Build and configure a production-ready FastAPI platform service behind Nginx to serve as the administrative CMDB control plane. The service dynamically routes queries to schema-isolated tenant databases via database search path transactions, authenticates administrators via User Login JWTs containing tenant slugs, provisions new clients via direct database migrations, provides administrative brokering for Sentinel join tokens, lists security telemetry via partition-pruned cursor pagination, and reactively transitions offline Sentinel agent statuses via an async Redis keyspace notification listener.

import jwt
from passlib.context import CryptContext
from fastapi import HTTPException, Security
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from app.core.config import settings

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")
security_bearer = HTTPBearer()

def verify_password(plain_password: str, hashed_password: str) -> bool:
    return pwd_context.verify(plain_password, hashed_password)

def get_password_hash(password: str) -> str:
    return pwd_context.hash(password)

def verify_user_jwt(credentials: HTTPAuthorizationCredentials = Security(security_bearer)) -> dict:
    """
    Decodes and verifies an administrative login User JWT.
    Validates expiration, signatures, and company claims.
    """
    try:
        token = credentials.credentials
        payload = jwt.decode(token, settings.JWT_SECRET, algorithms=[settings.JWT_ALGORITHM])
        
        # Ensure company_slug is embedded in claims
        if "company_slug" not in payload:
            raise HTTPException(status_code=401, detail="Missing company_slug claim in JWT payload")
        
        return payload
    except jwt.ExpiredSignatureError:
        raise HTTPException(status_code=401, detail="Administrative JWT signature has expired")
    except jwt.InvalidTokenError as exc:
        raise HTTPException(status_code=401, detail=f"Invalid user administrative token: {exc}")

def mint_sentinel_join_token(client_id: str, server_id: str) -> str:
    """
    Generates a signed one-time-use Join Token for a Sentinel agent.
    Signs the payload with BOOTSTRAP_JWT_SECRET.
    """
    import uuid
    now = int(time.time())
    payload = {
        "sub": "sentinel-bootstrap",
        "client_id": client_id,
        "server_id": server_id,
        "jti": str(uuid.uuid4()),
        "iat": now,
        "exp": now + 900  # 15 minutes TTL
    }
    return jwt.encode(payload, settings.BOOTSTRAP_JWT_SECRET, algorithm=settings.JWT_ALGORITHM)

from sqlalchemy import text
from pydantic import BaseModel, EmailStr
import jwt
import time
from app.core.database import get_db_session
from app.core.security import verify_password
from app.core.config import settings

router = APIRouter()

class LoginRequest(BaseModel):
    email: EmailStr
    password: str

class TokenResponse(BaseModel):
    access_token: str
    token_type: str

@router.post("/login", response_model=TokenResponse)
async def login(req: LoginRequest, db: AsyncSession = Depends(get_db_session)):
    """
    Authenticates a user against public.admin_users and retrieves their global
    role mapping or tenant company slug. Returns an active JWT token.
    """
    result = await db.execute(
        text("SELECT id, email, display_name, password_hash FROM public.admin_users WHERE email = :email AND status = 'active'"),
        {"email": req.email}
    )
    user = result.fetchone()
    if not user or not verify_password(req.password, user.password_hash):
        raise HTTPException(status_code=401, detail="Invalid admin credentials")
    
    # Query corresponding company slug
    tenant_result = await db.execute(
        text("""
            SELECT t.slug 
            FROM public.admin_user_roles aur
            JOIN public.tenants t ON t.id = aur.tenant_id
            WHERE aur.admin_user_id = :user_id LIMIT 1
        """),
        {"user_id": user.id}
    )
    tenant_row = tenant_result.fetchone()
    company_slug = tenant_row.slug if tenant_row else "global"

    payload = {
        "sub": str(user.id),
        "email": user.email,
        "name": user.display_name,
        "company_slug": company_slug,
        "iat": int(time.time()),
        "exp": int(time.time()) + (settings.ACCESS_TOKEN_EXPIRE_MINUTES * 60)
    }
    
    token = jwt.encode(payload, settings.JWT_SECRET, algorithm=settings.JWT_ALGORITHM)
    return TokenResponse(access_token=token, token_type="bearer")

import base64
from datetime import datetime, timezone
from typing import Optional
from fastapi import APIRouter, Depends, HTTPException, Query
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import text
from pydantic import BaseModel
from app.core.database import get_db_session, set_tenant_search_path
from app.core.security import verify_user_jwt

router = APIRouter()

class AlertPayload(BaseModel):
    id: str
    server_id: str
    event_type: str
    severity: str
    details: dict
    detected_at: str

class PagedAlertsResponse(BaseModel):
    data: list[AlertPayload]
    next_cursor: Optional[str]
    has_more: bool

@router.get("/alerts", response_model=PagedAlertsResponse)
async def get_alerts(
    cursor: Optional[str] = Query(None),
    limit: int = Query(50, ge=1, le=100),
    db: AsyncSession = Depends(get_db_session),
    admin_claims: dict = Depends(verify_user_jwt)
):
    """
    Exposes high-speed, partition-pruned cursor-paginated security alerts.
    Pruning is accomplished by forcing range comparisons directly on detected_at.
    """
    company_slug = admin_claims["company_slug"]
    if company_slug == "global":
        raise HTTPException(status_code=400, detail="Scope mismatch: Endpoint requires tenant authorization context")
    
    await set_tenant_search_path(db, company_slug)

    # Defaults: query the last 30 days of data to trigger partition pruning
    pruning_limit = "AND detected_at >= (now() - INTERVAL '30 days')"
    cursor_timestamp = None
    cursor_id = None

    # Parse cursor if present
    if cursor:
        try:
            decoded = base64.b64decode(cursor.encode("utf-8")).decode("utf-8")
            # Expected format: "2026-05-17T22:15:00+00:00||some-uuid-id"
            ts_str, cid = decoded.split("||")
            cursor_timestamp = datetime.fromisoformat(ts_str)
            cursor_id = cid
            # When paginate, search backwards from cursor point (newer to older)
            pruning_limit = "AND detected_at <= :cursor_ts AND (detected_at < :cursor_ts OR id < :cursor_id)"
        except Exception:
            raise HTTPException(status_code=400, detail="Invalid cursor payload mapping")

    # Dynamic cursor query execution
    # Prunes unused monthly tables by directly searching partition key detected_at
    sql = f"""
        SELECT id, server_id, event_type, severity, details, detected_at::text
        FROM tamper_alerts
        WHERE 1=1
        {pruning_limit}
        ORDER BY detected_at DESC, id DESC
        LIMIT :limit_plus_one
    """

    params = {"limit_plus_one": limit + 1}
    if cursor_timestamp:
        params["cursor_ts"] = cursor_timestamp
        params["cursor_id"] = cursor_id

    result = await db.execute(text(sql), params)
    rows = result.fetchall()

    has_more = len(rows) > limit
    data_rows = rows[:limit]

    next_cursor = None
    if has_more and data_rows:
        last_row = data_rows[-1]
        raw_cursor = f"{last_row.detected_at}||{last_row.id}"
        next_cursor = base64.b64encode(raw_cursor.encode("utf-8")).decode("utf-8")

    alerts_list = [
        AlertPayload(
            id=str(r.id),
            server_id=str(r.server_id),
            event_type=r.event_type,
            severity=r.severity,
            details=r.details,
            detected_at=r.detected_at
        ) for r in data_rows
    ]

    return PagedAlertsResponse(data=alerts_list, next_cursor=next_cursor, has_more=has_more)
```

import asyncio
import logging
import redis.asyncio as redis
from sqlalchemy.ext.asyncio import AsyncSession
from sqlalchemy import text
from app.core.config import settings
from app.core.database import async_session_factory

logger = logging.getLogger("platform.liveness")

async def listen_redis_heartbeat_expirations():
    """
    Subscribes to Redis keyspace expirations (__keyevent@0__:expired).
    Extracts agent_id and commits an async database transaction mapping
    that Sentinel status to 'offline' in its target schema.
    """
    logger.info("Initializing Redis keyspace expiration listener...")
    r = redis.from_url(settings.REDIS_URL, decode_responses=True)
    pubsub = r.pubsub()
    
    # Subscribe to expired events. Ensure Redis config notify-keyspace-events Ex is active.
    await pubsub.subscribe("__keyevent@0__:expired")
    
    try:
        async for message in pubsub.listen():
            if message["type"] != "message":
                continue
            
            expired_key = message["data"]
            # Expiration keys match: "agent:heartbeat:{company_slug}:{agent_id}"
            if not expired_key.startswith("agent:heartbeat:"):
                continue
            
            logger.warning(f"LIVENESS TIMEOUT DETECTED: Redis heartbeat key expired: {expired_key}")
            parts = expired_key.split(":")
            if len(parts) < 4:
                continue
            
            company_slug = parts[2]
            agent_id = parts[3]
            
            # Reactively write to DB to flag as offline
            await mark_agent_offline(company_slug, agent_id)
            
    except asyncio.CancelledError:
        logger.info("Keyspace subscription listener cancelled")
    except Exception as exc:
        logger.error(f"Keyspace listener error: {exc}")
        # Soft restart listener
        await asyncio.sleep(5)
        asyncio.create_task(listen_redis_heartbeat_expirations())

async def mark_agent_offline(company_slug: str, agent_id: str):
    """Dynamically logs session connections, sets search path, and updates status."""
    async with async_session_factory() as session:
        try:
            # Set search path to target client schema
            await session.execute(
                text("SELECT public.set_tenant_search_path(:slug)"),
                {"slug": company_slug}
            )
            
            # Reactively change active state to 'offline'
            up_res = await session.execute(
                text("""
                    UPDATE sentinel_agents 
                    SET status = 'offline', updated_at = now() 
                    WHERE id = :agent_id AND status != 'offline'
                """),
                {"agent_id": agent_id}
            )
            
            # Log control-plane state change
            if up_res.rowcount > 0:
                # Add to public global audit log
                tenant_res = await session.execute(
                    text("SELECT id FROM public.tenants WHERE slug = :slug"),
                    {"slug": company_slug}
                )
                tenant_row = tenant_res.fetchone()
                if tenant_row:
                    await session.execute(
                        text("""
                            INSERT INTO public.global_audit_log (tenant_id, event_type, payload)
                            VALUES (:tenant_id, 'SENTINEL_LIVENESS_EXPIRED', :payload)
                        """),
                        {
                            "tenant_id": tenant_row.id,
                            "payload": '{"agent_id": "%s"}' % agent_id
                        }
                    )
                
                await session.commit()
                logger.warning(f"SENTINEL_OFFLINE: Agent {agent_id} for client {company_slug} marked offline.")
            
        except Exception as exc:
            await session.rollback()
            logger.error(f"Failed to record offline transition for agent {agent_id} in {company_slug}: {exc}")
```
